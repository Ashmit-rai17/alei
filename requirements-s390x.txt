# Pinned to the s390x (LinuxONE) demo box. Train the final model in .venv-ship with exactly these.
#
# Why these versions: scikit-learn has no s390x wheel above 1.4; 1.9.1 falls back to a source build
# that needs gfortran, which the target does not have. The target runs distro-provided versions, so
# we match them here. A .joblib only loads reliably under the scikit-learn version that wrote it.
#
#   python3.12 -m venv .venv-ship
#   .venv-ship/bin/pip install --upgrade pip
#   .venv-ship/bin/pip install -r requirements-s390x.txt
#   .venv-ship/bin/python -c "import numpy,sklearn,scipy,joblib;print(numpy.__version__,sklearn.__version__,scipy.__version__,joblib.__version__)"
#   -> 1.26.4 1.4.1.post1 1.11.4 1.3.2
#
# predict.py on s390x needs only the first four (plus threadpoolctl, pulled in by scikit-learn).

numpy==1.26.4
scipy==1.11.4
scikit-learn==1.4.1.post1
joblib==1.3.2

# training only (not needed on s390x)
pandas<2.2            # resolves to 2.1.4
pyarrow==17.0.0       # newer pyarrow requires NumPy >= 2.0 and fails to import under numpy 1.26.4
openpyxl              # 3.1.5
